Option Explicit

' ============================================================
' Excel側 軽量制御モジュール
'
' Excel/VBAは以下のみ担当:
' - README作成
' - 入力シート整形
' - Sheet2 A2以降のハイパーリンクを消さずに整理
' - ジョブTSV出力
' - BAT起動
' - 結果TSVの取込
'
' 重い処理:
' - ダウンロード
' - ZIP展開
' - PDF化
' - HTML化
' - HTML結合
'
' は run_external_downloader.bat + external_downloader.ps1 側で実行する。
' ============================================================

Private Const SETTINGS_SHEET_NAME As String = "01_SETTINGS"
Private Const LINKS_SHEET_NAME As String = "02_LINKS"
Private Const README_SHEET_NAME As String = "README"

Private Const FIRST_LINK_ROW As Long = 2

Private Const DIR_ZIP As String = "ZIP"
Private Const DIR_EXTRACT As String = "EXTRACT"
Private Const DIR_PDF As String = "PDF"
Private Const DIR_HTML As String = "HTML"
Private Const DIR_HTML_COMBINE As String = "HTML_COMBINE"
Private Const DIR_WORK As String = "WORK"

Private Const BAT_FILE_NAME As String = "run_external_downloader.bat"
Private Const PS1_FILE_NAME As String = "external_downloader.ps1"

Private Const DEFAULT_WORKER_COUNT As Long = 3
Private Const MAX_WORKER_COUNT As Long = 8

' ============================================================
' 初期セットアップ
' ============================================================

Public Sub SetupExternalDownloaderWorkbook()

    Dim wsCfg As Worksheet
    Dim wsLinks As Worksheet
    Dim wsReadme As Worksheet

    On Error GoTo ErrHandler

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    Set wsCfg = EnsureSheetByPosition(1, SETTINGS_SHEET_NAME)
    Set wsLinks = EnsureSheetByPosition(2, LINKS_SHEET_NAME)
    Set wsReadme = EnsureSheetByName(README_SHEET_NAME)

    BuildSettingsSheet wsCfg
    BuildLinksSheetPreserveData wsLinks
    BuildReadmeSheet wsReadme

    wsCfg.Activate

    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox _
        "セットアップが完了しました。" & vbCrLf & _
        "01_SETTINGS の A1〜A3 と、02_LINKS の A2以降を入力してください。", _
        vbInformation

    Exit Sub

ErrHandler:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "セットアップ中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' 外部実行
' ============================================================

Public Sub LaunchExternalDownloadExtractPdfHtml()

    Dim wsCfg As Worksheet
    Dim wsLinks As Worksheet

    Dim proxyRaw As String
    Dim rootDir As String
    Dim workerCount As Long

    Dim batPath As String
    Dim ps1Path As String
    Dim jobPath As String

    Dim cmd As String
    Dim sh As Object

    On Error GoTo ErrHandler

    Set wsCfg = ThisWorkbook.Worksheets(1)
    Set wsLinks = ThisWorkbook.Worksheets(2)

    proxyRaw = Trim(CStr(wsCfg.Range("A1").Value))
    rootDir = NormalizeFolderPath(CStr(wsCfg.Range("A2").Value))
    workerCount = CLng(Val(CStr(wsCfg.Range("A3").Value)))

    If workerCount <= 0 Then workerCount = DEFAULT_WORKER_COUNT
    If workerCount > MAX_WORKER_COUNT Then workerCount = MAX_WORKER_COUNT

    If Len(rootDir) = 0 Then
        MsgBox "01_SETTINGS!A2 に出力先ルートフォルダを入力してください。", vbExclamation
        Exit Sub
    End If

    EnsureFolderExists rootDir
    EnsureFolderExists CombinePath(rootDir, DIR_ZIP)
    EnsureFolderExists CombinePath(rootDir, DIR_EXTRACT)
    EnsureFolderExists CombinePath(rootDir, DIR_PDF)
    EnsureFolderExists CombinePath(rootDir, DIR_HTML)
    EnsureFolderExists CombinePath(rootDir, DIR_HTML_COMBINE)
    EnsureFolderExists CombinePath(rootDir, DIR_WORK)

    batPath = ResolveSupportFilePath(rootDir, BAT_FILE_NAME)
    ps1Path = ResolveSupportFilePath(rootDir, PS1_FILE_NAME)

    If Len(batPath) = 0 Then
        MsgBox _
            BAT_FILE_NAME & " が見つかりません。" & vbCrLf & _
            "Excelファイルと同じフォルダ、または出力先ルートフォルダに保存してください。", _
            vbExclamation
        Exit Sub
    End If

    If Len(ps1Path) = 0 Then
        MsgBox _
            PS1_FILE_NAME & " が見つかりません。" & vbCrLf & _
            "Excelファイルと同じフォルダ、または出力先ルートフォルダに保存してください。", _
            vbExclamation
        Exit Sub
    End If

    ' リンクを消さずに整形する
    CleanLinkSheetPreserveHyperlinks False

    jobPath = ExportJobTsv(wsLinks, rootDir)

    If Len(jobPath) = 0 Then
        MsgBox _
            "処理対象URLがありません。" & vbCrLf & _
            "02_LINKS の A2以降にハイパーリンクまたはURL文字列を入力してください。", _
            vbExclamation
        Exit Sub
    End If

    wsCfg.Range("A5").Value = jobPath
    wsCfg.Range("B5").Value = "直近ジョブTSV"

    cmd = "cmd.exe /k " & _
          QuoteArg(batPath) & " " & _
          QuoteArg(jobPath) & " " & _
          QuoteArg(rootDir) & " " & _
          QuoteArg(proxyRaw) & " " & _
          QuoteArg(CStr(workerCount))

    Set sh = CreateObject("WScript.Shell")

    Debug.Print Now & " [EXCEL] 外部実行開始"
    Debug.Print Now & " [EXCEL] " & cmd

    ' 非同期起動。Excelは固まらない。
    sh.Run cmd, 1, False

    MsgBox _
        "外部処理を起動しました。" & vbCrLf & _
        "進捗は起動した黒い画面に表示されます。" & vbCrLf & _
        "完了後、必要に応じて ImportLatestExternalResults を実行してください。", _
        vbInformation

    Exit Sub

ErrHandler:
    MsgBox "外部実行の起動中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' 結果取込
' ============================================================

Public Sub ImportLatestExternalResults()

    Dim wsCfg As Worksheet
    Dim wsLinks As Worksheet

    Dim rootDir As String
    Dim workDir As String
    Dim latestRunDir As String

    Dim imported As Long

    On Error GoTo ErrHandler

    Set wsCfg = ThisWorkbook.Worksheets(1)
    Set wsLinks = ThisWorkbook.Worksheets(2)

    rootDir = NormalizeFolderPath(CStr(wsCfg.Range("A2").Value))

    If Len(rootDir) = 0 Then
        MsgBox "01_SETTINGS!A2 に出力先ルートフォルダを入力してください。", vbExclamation
        Exit Sub
    End If

    workDir = CombinePath(rootDir, DIR_WORK)
    latestRunDir = FindLatestRunFolder(workDir)

    If Len(latestRunDir) = 0 Then
        MsgBox "WORKフォルダ内に run_ から始まる実行結果フォルダが見つかりません。", vbExclamation
        Exit Sub
    End If

    imported = ImportResultsFromRunDir(wsLinks, latestRunDir)

    MsgBox _
        "結果を取り込みました。" & vbCrLf & _
        "取込件数: " & imported & vbCrLf & _
        "対象フォルダ: " & latestRunDir, _
        vbInformation

    Exit Sub

ErrHandler:
    MsgBox "結果取込中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' Sheet2 A2以降のリンクを消さずに整形
' ============================================================

Public Sub CleanLinkSheetPreserveHyperlinks(Optional ByVal showMessage As Boolean = True)

    Dim wsLinks As Worksheet
    Dim lastRow As Long
    Dim r As Long

    Dim displayText As String
    Dim address As String
    Dim subAddress As String

    Dim c As Range

    On Error GoTo ErrHandler

    Set wsLinks = ThisWorkbook.Worksheets(2)

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    lastRow = wsLinks.Cells(wsLinks.Rows.Count, "A").End(xlUp).Row

    If lastRow < FIRST_LINK_ROW Then
        BuildLinksSheetPreserveData wsLinks
        GoTo Finally
    End If

    For r = FIRST_LINK_ROW To lastRow

        Set c = wsLinks.Cells(r, "A")

        displayText = ""
        address = ""
        subAddress = ""

        GetCellTextAndHyperlink c, displayText, address, subAddress

        If Len(displayText) = 0 And Len(address) > 0 Then
            displayText = address
        End If

        If Len(address) = 0 And IsHttpUrl(displayText) Then
            address = displayText
        End If

        ' ここでセル自体は消さず、内容とリンクを作り直す
        c.Hyperlinks.Delete
        c.ClearFormats
        c.NumberFormat = "@"

        If Len(displayText) > 0 Or Len(address) > 0 Or Len(subAddress) > 0 Then

            c.Value = displayText

            If Len(address) > 0 Or Len(subAddress) > 0 Then
                wsLinks.Hyperlinks.Add _
                    Anchor:=c, _
                    Address:=address, _
                    SubAddress:=subAddress, _
                    TextToDisplay:=displayText
            End If

        Else
            c.ClearContents
        End If

    Next r

    BuildLinksSheetPreserveData wsLinks

Finally:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    If showMessage Then
        MsgBox "02_LINKS A2以降を、テキストとハイパーリンクのみ保持する形に整理しました。", vbInformation
    End If

    Exit Sub

ErrHandler:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "リンク整理中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' 外部更新リンク解除
' ============================================================

Public Sub BreakExternalUpdateLinksOnly()

    Dim links As Variant
    Dim i As Long
    Dim count As Long

    On Error GoTo ErrHandler

    links = ThisWorkbook.LinkSources(Type:=xlLinkTypeExcelLinks)

    If IsEmpty(links) Then
        MsgBox "外部更新リンクは見つかりませんでした。", vbInformation
        Exit Sub
    End If

    For i = LBound(links) To UBound(links)
        ThisWorkbook.BreakLink _
            Name:=CStr(links(i)), _
            Type:=xlLinkTypeExcelLinks
        count = count + 1
    Next i

    MsgBox "外部更新リンクを " & count & " 件解除しました。", vbInformation

    Exit Sub

ErrHandler:
    MsgBox "外部更新リンク解除中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' ジョブTSV出力
' ============================================================

Private Function ExportJobTsv( _
        ByVal wsLinks As Worksheet, _
        ByVal rootDir As String _
    ) As String

    Dim workDir As String
    Dim jobPath As String
    Dim lastRow As Long
    Dim r As Long

    Dim displayText As String
    Dim address As String
    Dim subAddress As String
    Dim url As String

    Dim lines As String
    Dim count As Long

    workDir = CombinePath(rootDir, DIR_WORK)
    EnsureFolderExists workDir

    lastRow = wsLinks.Cells(wsLinks.Rows.Count, "A").End(xlUp).Row

    For r = FIRST_LINK_ROW To lastRow

        displayText = ""
        address = ""
        subAddress = ""

        GetCellTextAndHyperlink wsLinks.Cells(r, "A"), displayText, address, subAddress

        url = address

        If Len(url) = 0 And IsHttpUrl(displayText) Then
            url = displayText
        End If

        If IsHttpUrl(url) Then
            count = count + 1
            lines = lines & CStr(r) & vbTab & CleanTsvField(url) & vbCrLf
            wsLinks.Cells(r, "B").Value = url
            wsLinks.Cells(r, "G").Value = "外部処理待機"
        End If

    Next r

    If count = 0 Then
        ExportJobTsv = ""
        Exit Function
    End If

    jobPath = CombinePath(workDir, "jobs_from_excel_" & Format(Now, "yyyymmdd_hhnnss") & ".tsv")
    WriteTextUtf8 jobPath, lines

    ExportJobTsv = jobPath

End Function

Private Function FindLatestRunFolder(ByVal workDir As String) As String

    Dim fso As Object
    Dim folder As Object
    Dim subFolder As Object

    Dim latestPath As String
    Dim latestDate As Date

    Set fso = CreateObject("Scripting.FileSystemObject")

    If Not fso.FolderExists(workDir) Then Exit Function

    Set folder = fso.GetFolder(workDir)

    For Each subFolder In folder.SubFolders

        If LCase(Left(CStr(subFolder.Name), 4)) = "run_" Then

            If Len(latestPath) = 0 Or subFolder.DateLastModified > latestDate Then
                latestPath = CStr(subFolder.Path)
                latestDate = subFolder.DateLastModified
            End If

        End If

    Next subFolder

    FindLatestRunFolder = latestPath

End Function

Private Function ImportResultsFromRunDir( _
        ByVal wsLinks As Worksheet, _
        ByVal runDir As String _
    ) As Long

    Dim fso As Object
    Dim folder As Object
    Dim file As Object

    Dim txt As String
    Dim lines() As String
    Dim parts() As String
    Dim line As String
    Dim i As Long
    Dim rowNo As Long

    Set fso = CreateObject("Scripting.FileSystemObject")

    If Not fso.FolderExists(runDir) Then Exit Function

    Set folder = fso.GetFolder(runDir)

    For Each file In folder.Files

        If LCase(Left(CStr(file.Name), 7)) = "result_" And LCase(Right(CStr(file.Name), 4)) = ".tsv" Then

            txt = ReadTextUtf8(CStr(file.Path))
            txt = Replace(txt, vbCrLf, vbLf)
            txt = Replace(txt, vbCr, vbLf)

            lines = Split(txt, vbLf)

            For i = LBound(lines) To UBound(lines)

                line = Trim(lines(i))

                If Len(line) > 0 Then

                    parts = Split(line, vbTab)

                    If UBound(parts) >= 6 Then

                        rowNo = CLng(Replace(parts(0), ChrW(&HFEFF), ""))

                        wsLinks.Cells(rowNo, "B").Value = parts(1)
                        wsLinks.Cells(rowNo, "C").Value = parts(2)
                        wsLinks.Cells(rowNo, "D").Value = parts(3)
                        wsLinks.Cells(rowNo, "E").Value = parts(4)
                        wsLinks.Cells(rowNo, "F").Value = parts(5)
                        wsLinks.Cells(rowNo, "G").Value = parts(6)

                        ImportResultsFromRunDir = ImportResultsFromRunDir + 1

                    End If

                End If

            Next i

        End If

    Next file

End Function

Private Sub GetCellTextAndHyperlink( _
        ByVal cell As Range, _
        ByRef displayText As String, _
        ByRef address As String, _
        ByRef subAddress As String _
    )

    On Error Resume Next

    displayText = ""

    If IsError(cell.Value) Then
        displayText = ""
    Else
        displayText = Trim(CStr(cell.Value))
    End If

    address = ""
    subAddress = ""

    If cell.Hyperlinks.Count > 0 Then

        address = CStr(cell.Hyperlinks(1).Address)
        subAddress = CStr(cell.Hyperlinks(1).SubAddress)

        If Len(displayText) = 0 Then
            displayText = CStr(cell.Hyperlinks(1).TextToDisplay)
        End If

    End If

    If Len(address) = 0 And cell.HasFormula Then
        TryParseHyperlinkFormula CStr(cell.Formula), address, displayText
    End If

    On Error GoTo 0

End Sub

Private Sub TryParseHyperlinkFormula( _
        ByVal formulaText As String, _
        ByRef address As String, _
        ByRef displayText As String _
    )

    Dim re As Object
    Dim matches As Object

    On Error GoTo ErrHandler

    If InStr(1, formulaText, "HYPERLINK", vbTextCompare) = 0 Then Exit Sub

    Set re = CreateObject("VBScript.RegExp")

    With re
        .Global = True
        .IgnoreCase = True
        .Pattern = """([^""]*)"""
    End With

    Set matches = re.Execute(formulaText)

    If matches.Count >= 1 Then

        address = CStr(matches(0).SubMatches(0))

        If matches.Count >= 2 Then
            displayText = CStr(matches(1).SubMatches(0))
        ElseIf Len(displayText) = 0 Then
            displayText = address
        End If

    End If

    Exit Sub

ErrHandler:
End Sub

Private Function EnsureSheetByPosition( _
        ByVal indexNo As Long, _
        ByVal sheetName As String _
    ) As Worksheet

    Dim ws As Worksheet

    Do While ThisWorkbook.Worksheets.Count < indexNo
        ThisWorkbook.Worksheets.Add After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count)
    Loop

    Set ws = ThisWorkbook.Worksheets(indexNo)

    On Error Resume Next
    ws.Name = sheetName
    On Error GoTo 0

    Set EnsureSheetByPosition = ws

End Function

Private Function EnsureSheetByName(ByVal sheetName As String) As Worksheet

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        ws.Name = sheetName
    End If

    Set EnsureSheetByName = ws

End Function

Private Function ResolveSupportFilePath( _
        ByVal rootDir As String, _
        ByVal fileName As String _
    ) As String

    Dim p As String
    Dim fso As Object

    Set fso = CreateObject("Scripting.FileSystemObject")

    If Len(ThisWorkbook.Path) > 0 Then
        p = CombinePath(ThisWorkbook.Path, fileName)
        If fso.FileExists(p) Then
            ResolveSupportFilePath = p
            Exit Function
        End If
    End If

    p = CombinePath(rootDir, fileName)
    If fso.FileExists(p) Then
        ResolveSupportFilePath = p
        Exit Function
    End If

    ResolveSupportFilePath = ""

End Function

Private Function NormalizeFolderPath(ByVal folderPath As String) As String

    folderPath = Trim(folderPath)
    folderPath = Replace(folderPath, """", "")

    Do While Len(folderPath) > 3 And _
        (Right(folderPath, 1) = "\" Or Right(folderPath, 1) = "/")
        folderPath = Left(folderPath, Len(folderPath) - 1)
    Loop

    NormalizeFolderPath = folderPath

End Function

Private Function CombinePath( _
        ByVal folderPath As String, _
        ByVal fileName As String _
    ) As String

    If Right(folderPath, 1) = "\" Or Right(folderPath, 1) = "/" Then
        CombinePath = folderPath & fileName
    Else
        CombinePath = folderPath & "\" & fileName
    End If

End Function

Private Sub EnsureFolderExists(ByVal folderPath As String)

    Dim fso As Object
    Dim parentPath As String

    Set fso = CreateObject("Scripting.FileSystemObject")

    folderPath = NormalizeFolderPath(folderPath)

    If Len(folderPath) = 0 Then Exit Sub
    If fso.FolderExists(folderPath) Then Exit Sub

    parentPath = fso.GetParentFolderName(folderPath)

    If Len(parentPath) > 0 Then
        If Not fso.FolderExists(parentPath) Then
            EnsureFolderExists parentPath
        End If
    End If

    If Not fso.FolderExists(folderPath) Then
        fso.CreateFolder folderPath
    End If

End Sub

Private Sub WriteTextUtf8( _
        ByVal filePath As String, _
        ByVal text As String _
    )

    Dim stream As Object

    Set stream = CreateObject("ADODB.Stream")

    With stream
        .Type = 2
        .Charset = "utf-8"
        .Open
        .WriteText text
        .SaveToFile filePath, 2
        .Close
    End With

End Sub

Private Function ReadTextUtf8(ByVal filePath As String) As String

    On Error GoTo ErrHandler

    Dim stream As Object

    Set stream = CreateObject("ADODB.Stream")

    With stream
        .Type = 2
        .Charset = "utf-8"
        .Open
        .LoadFromFile filePath
        ReadTextUtf8 = .ReadText
        .Close
    End With

    Exit Function

ErrHandler:
    ReadTextUtf8 = ""

End Function

Private Function CleanTsvField(ByVal s As String) As String

    s = Replace(s, vbTab, " ")
    s = Replace(s, vbCr, " ")
    s = Replace(s, vbLf, " ")

    CleanTsvField = s

End Function

Private Function IsHttpUrl(ByVal s As String) As Boolean

    Dim u As String

    u = LCase(Trim(s))

    IsHttpUrl = _
        Left(u, 7) = "http://" Or _
        Left(u, 8) = "https://"

End Function

Private Function QuoteArg(ByVal s As String) As String

    QuoteArg = """" & Replace(s, """", """""") & """"

End Function
