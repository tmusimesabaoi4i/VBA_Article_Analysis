Option Explicit

' ============================================================
' 外部処理版 Excel制御VBA
'
' 方針:
'   - Excel/VBAは軽量制御のみ
'   - 重い処理は BAT + PowerShell 側で実行
'   - 02_LINKS A列は原本列として絶対に壊さない
'   - A2以降のハイパーリンクは削除・再作成しない
'   - URL抽出結果だけをB列に出す
'
' 主な実行マクロ:
'   1. SetupExternalDownloaderWorkbook
'   2. RefreshLinkUrlColumn
'   3. LaunchExternalDownloadExtractPdfHtml
'   4. ImportLatestExternalResults
' ============================================================

Private Const SETTINGS_SHEET_NAME As String = "01_SETTINGS"
Private Const LINKS_SHEET_NAME As String = "02_LINKS"
Private Const README_SHEET_NAME As String = "README"

Private Const DIR_EXTRACT As String = "EXTRACT"

Private Const DIR_HTML As String = "HTML"
Private Const DIR_HTML_COMBINE As String = "HTML_COMBINE"
Private Const DIR_WORK As String = "WORK"

Private Const BAT_FILE_NAME As String = "run_external_downloader.bat"

' ============================================================
' 初期セットアップ
' ============================================================

Public Sub SetupExternalDownloaderWorkbook()

    Dim wsCfg As Worksheet
    Dim wsLinks As Worksheet
    Dim wsReadme As Worksheet

    On Error GoTo ErrHandler

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    Set wsCfg = EnsureSheetByPosition(1, SETTINGS_SHEET_NAME)
    Set wsLinks = EnsureSheetByPosition(2, LINKS_SHEET_NAME)
    Set wsReadme = EnsureSheetByName(README_SHEET_NAME)

    BuildSettingsSheet wsCfg
    BuildLinksSheetNonDestructive wsLinks
    BuildReadmeSheet wsReadme

    wsCfg.Activate

    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox _
        "セットアップが完了しました。" & vbCrLf & _
        "01_SETTINGS の A1〜A3 と、02_LINKS の A2以降を入力してください。" & vbCrLf & _
        "02_LINKS のA列は原本列として保持されます。", _
        vbInformation

    Exit Sub

ErrHandler:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "セットアップ中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' リンクURL抽出
'
' 重要:
'   A列は消さない
'   A列のHyperlinkを削除しない
'   A列のHyperlinkを再作成しない
'   抽出URLのみB列に表示
' ============================================================

Public Sub RefreshLinkUrlColumn(Optional ByVal showMessage As Boolean = True)

    Dim wsLinks As Worksheet
    Dim lastRow As Long
    Dim r As Long
    Dim url As String
    Dim label As String

    On Error GoTo ErrHandler

    Set wsLinks = ThisWorkbook.Worksheets(2)

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    BuildLinksSheetNonDestructive wsLinks

    lastRow = GetLastUsedRowInColumn(wsLinks, "A")

    If lastRow < FIRST_LINK_ROW Then
        GoTo Finally
    End If

    ' B:Gは結果列なのでクリアしてよい
    wsLinks.Range("B" & FIRST_LINK_ROW & ":G" & lastRow).ClearContents
    wsLinks.Range("B" & FIRST_LINK_ROW & ":G" & lastRow).ClearFormats

    For r = FIRST_LINK_ROW To lastRow

        url = ResolveDownloadUrlFromCell(wsLinks.Cells(r, "A"))
        label = SafeCellText(wsLinks.Cells(r, "A"))

        If Len(url) > 0 Then
            wsLinks.Cells(r, "B").Value = url
            wsLinks.Cells(r, "G").Value = "URL抽出OK"
        Else
            If Len(label) > 0 Then
                wsLinks.Cells(r, "G").Value = "URL抽出不可"
            End If
        End If

        StyleLinkInputRow wsLinks, r

    Next r

    StyleResultColumns wsLinks, lastRow

Finally:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    If showMessage Then
        MsgBox _
            "URL抽出が完了しました。" & vbCrLf & _
            "A列のリンク・表示文字列は変更していません。" & vbCrLf & _
            "抽出URLはB列に表示しています。", _
            vbInformation
    End If

    Exit Sub

ErrHandler:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox _
        "URL抽出中にエラーが発生しました。" & vbCrLf & _
        "行: " & r & vbCrLf & _
        "内容: " & Err.Description, _
        vbExclamation

End Sub

End Sub

' ============================================================
' 外部処理起動
' ============================================================

Public Sub LaunchExternalDownloadExtractPdfHtml()

    Dim wsCfg As Worksheet
    Dim wsLinks As Worksheet

    Dim proxyRaw As String
    Dim rootDir As String
    Dim workerCount As Long

    Dim batPath As String
    Dim ps1Path As String
    Dim jobPath As String

    Dim cmd As String
    Dim sh As Object

    On Error GoTo ErrHandler

    Set wsCfg = ThisWorkbook.Worksheets(1)
    Set wsLinks = ThisWorkbook.Worksheets(2)

    proxyRaw = Trim(CStr(wsCfg.Range("A1").Value))
    rootDir = NormalizeFolderPath(CStr(wsCfg.Range("A2").Value))
    workerCount = CLng(Val(CStr(wsCfg.Range("A3").Value)))

    If workerCount <= 0 Then workerCount = DEFAULT_WORKER_COUNT
    If workerCount > MAX_WORKER_COUNT Then workerCount = MAX_WORKER_COUNT

    If Len(rootDir) = 0 Then
        MsgBox "01_SETTINGS!A2 に出力先ルートフォルダを入力してください。", vbExclamation
        Exit Sub
    End If

    EnsureFolderExists rootDir
    EnsureFolderExists CombinePath(rootDir, DIR_ZIP)
    EnsureFolderExists CombinePath(rootDir, DIR_EXTRACT)
    EnsureFolderExists CombinePath(rootDir, DIR_PDF)
    EnsureFolderExists CombinePath(rootDir, DIR_HTML)
    EnsureFolderExists CombinePath(rootDir, DIR_HTML_COMBINE)
    EnsureFolderExists CombinePath(rootDir, DIR_WORK)

    batPath = ResolveSupportFilePath(rootDir, BAT_FILE_NAME)
    ps1Path = ResolveSupportFilePath(rootDir, PS1_FILE_NAME)

    If Len(batPath) = 0 Then
        MsgBox _
            BAT_FILE_NAME & " が見つかりません。" & vbCrLf & _
            "Excelファイルと同じフォルダ、または出力先ルートフォルダに保存してください。", _
            vbExclamation
        Exit Sub
    End If

    If Len(ps1Path) = 0 Then
        MsgBox _
            PS1_FILE_NAME & " が見つかりません。" & vbCrLf & _
            "Excelファイルと同じフォルダ、または出力先ルートフォルダに保存してください。", _
            vbExclamation
        Exit Sub
    End If

    ' A列を壊さず、B列にURLだけ抽出
    RefreshLinkUrlColumn False

    jobPath = ExportJobTsv(wsLinks, rootDir)

    If Len(jobPath) = 0 Then
        MsgBox _
            "処理対象URLがありません。" & vbCrLf & _
            "02_LINKS の A2以降に、ハイパーリンクまたはURL文字列を入力してください。" & vbCrLf & _
            "B列にURLが抽出されているかも確認してください。", _
            vbExclamation
        Exit Sub
    End If

    wsCfg.Range("A5").Value = jobPath
    wsCfg.Range("B5").Value = "直近ジョブTSV"

    cmd = "cmd.exe /k " & _
          QuoteArg(batPath) & " " & _
          QuoteArg(jobPath) & " " & _
          QuoteArg(rootDir) & " " & _
          QuoteArg(proxyRaw) & " " & _
          QuoteArg(CStr(workerCount))

    Set sh = CreateObject("WScript.Shell")

    Debug.Print Now & " [EXCEL] 外部実行開始"
    Debug.Print Now & " [EXCEL] " & cmd

    ' 非同期起動。Excelは固まらない。
    sh.Run cmd, 1, False

    MsgBox _
        "外部処理を起動しました。" & vbCrLf & _
        "進捗は起動した黒い画面に表示されます。" & vbCrLf & _
        "完了後、ImportLatestExternalResults を実行してください。", _
        vbInformation

    Exit Sub

ErrHandler:
    MsgBox "外部実行の起動中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================

Public Sub ImportLatestExternalResults()

    Dim wsCfg As Worksheet
    Dim wsLinks As Worksheet

    Dim rootDir As String
    Dim workDir As String
    Dim latestRunDir As String

    Dim imported As Long

    On Error GoTo ErrHandler

    Set wsCfg = ThisWorkbook.Worksheets(1)
    Set wsLinks = ThisWorkbook.Worksheets(2)

    rootDir = NormalizeFolderPath(CStr(wsCfg.Range("A2").Value))

    If Len(rootDir) = 0 Then
        MsgBox "01_SETTINGS!A2 に出力先ルートフォルダを入力してください。", vbExclamation
        Exit Sub
    End If

    workDir = CombinePath(rootDir, DIR_WORK)
    latestRunDir = FindLatestRunFolder(workDir)

    If Len(latestRunDir) = 0 Then
        MsgBox "WORKフォルダ内に run_ から始まる実行結果フォルダが見つかりません。", vbExclamation
        Exit Sub
    End If

    imported = ImportResultsFromRunDir(wsLinks, latestRunDir)

    MsgBox _
        "結果を取り込みました。" & vbCrLf & _
        "取込件数: " & imported & vbCrLf & _
        "対象フォルダ: " & latestRunDir, _
        vbInformation

    Exit Sub

ErrHandler:
    MsgBox "結果取込中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

Public Sub ClearResultsOnly()

    Dim wsLinks As Worksheet
    Dim lastRow As Long

    On Error GoTo ErrHandler

    Set wsLinks = ThisWorkbook.Worksheets(2)

    lastRow = Application.WorksheetFunction.Max( _
                GetLastUsedRowInColumn(wsLinks, "A"), _
                GetLastUsedRowInColumn(wsLinks, "G") _
              )

    If lastRow < FIRST_LINK_ROW Then Exit Sub

    wsLinks.Range("B" & FIRST_LINK_ROW & ":G" & lastRow).ClearContents

    MsgBox "B〜G列の結果欄だけクリアしました。A列は変更していません。", vbInformation

    Exit Sub

ErrHandler:

End Sub

' ============================================================
' ジョブTSV出力
' ============================================================

Private Function ExportJobTsv( _
        ByVal wsLinks As Worksheet, _
        ByVal rootDir As String _
    ) As String

    Dim workDir As String
    Dim jobPath As String
    Dim lastRow As Long
    Dim r As Long
    Dim url As String
    Dim lines As String
    Dim count As Long

    workDir = CombinePath(rootDir, DIR_WORK)
    EnsureFolderExists workDir

    lastRow = GetLastUsedRowInColumn(wsLinks, "A")

    For r = FIRST_LINK_ROW To lastRow

        url = ResolveDownloadUrlFromCell(wsLinks.Cells(r, "A"))

        If IsHttpUrl(url) Then
            count = count + 1
            lines = lines & CStr(r) & vbTab & CleanTsvField(url) & vbCrLf
            wsLinks.Cells(r, "B").Value = url
            wsLinks.Cells(r, "G").Value = "外部処理待機"
        Else
            If Len(SafeCellText(wsLinks.Cells(r, "A"))) > 0 Then
                wsLinks.Cells(r, "G").Value = "URL対象外"
            End If
        End If

    Next r

    If count = 0 Then
        ExportJobTsv = ""
        Exit Function
    End If

    jobPath = CombinePath(workDir, "jobs_from_excel_" & Format(Now, "yyyymmdd_hhnnss") & ".tsv")
    WriteTextUtf8 jobPath, lines

    ExportJobTsv = jobPath

End Function

' ============================================================
' URL解決ロジック
'
' 優先順位:
'   1. セルのHyperlinks(1).Address
'   2. HYPERLINK関数内のURL
'   3. セル値そのものがURL
'   4. セル表示文字列中に含まれるURL
'
' 内部リンク、外部ブック参照、file://等は対象外
' ============================================================

Private Function ResolveDownloadUrlFromCell(ByVal cell As Range) As String

    Dim s As String
    Dim formulaText As String
    Dim valueText As String
    Dim textText As String

    On Error GoTo SafeExit

    ' 1. Excelハイパーリンクオブジェクト
    s = SafeHyperlinkAddress(cell)
    s = NormalizeUrlCandidate(s)

    If IsHttpUrl(s) Then
        ResolveDownloadUrlFromCell = s
        Exit Function
    End If

    ' 2. HYPERLINK関数
    formulaText = SafeCellFormula(cell)

    If Len(formulaText) > 0 Then
        s = ExtractFirstHttpUrl(formulaText)
        s = NormalizeUrlCandidate(s)

        If IsHttpUrl(s) Then
            ResolveDownloadUrlFromCell = s
            Exit Function
        End If
    End If

    ' 3. セル値
    valueText = SafeCellValueAsText(cell)
    s = NormalizeUrlCandidate(valueText)

    If IsHttpUrl(s) Then
        ResolveDownloadUrlFromCell = s
        Exit Function
    End If

    ' 4. 表示文字列中のURL
    textText = SafeCellText(cell)
    s = ExtractFirstHttpUrl(textText)
    s = NormalizeUrlCandidate(s)

    If IsHttpUrl(s) Then
        ResolveDownloadUrlFromCell = s
        Exit Function
    End If

SafeExit:
    ResolveDownloadUrlFromCell = ""

End Function

Private Function SafeHyperlinkAddress(ByVal cell As Range) As String

    On Error GoTo ErrHandler

    If cell.Hyperlinks.Count > 0 Then
        SafeHyperlinkAddress = CStr(cell.Hyperlinks(1).Address)
    Else
        SafeHyperlinkAddress = ""
    End If

    Exit Function

ErrHandler:
    SafeHyperlinkAddress = ""

End Function

Private Function SafeHyperlinkSubAddress(ByVal cell As Range) As String

    On Error GoTo ErrHandler

    If cell.Hyperlinks.Count > 0 Then
        SafeHyperlinkSubAddress = CStr(cell.Hyperlinks(1).SubAddress)
    Else
        SafeHyperlinkSubAddress = ""
    End If

    Exit Function

ErrHandler:
    SafeHyperlinkSubAddress = ""

End Function

Private Function SafeCellFormula(ByVal cell As Range) As String

    On Error GoTo ErrHandler

    If cell.HasFormula Then
        SafeCellFormula = CStr(cell.Formula)
    Else
        SafeCellFormula = ""
    End If

    Exit Function

ErrHandler:
    SafeCellFormula = ""

End Function

Private Function SafeCellValueAsText(ByVal cell As Range) As String

    On Error GoTo ErrHandler

    If IsError(cell.Value) Then
        SafeCellValueAsText = ""
    Else
        SafeCellValueAsText = Trim(CStr(cell.Value))
    End If

    Exit Function

ErrHandler:
    SafeCellValueAsText = ""

End Function

Private Function SafeCellText(ByVal cell As Range) As String

    On Error GoTo ErrHandler

    If IsError(cell.Value) Then
        SafeCellText = ""
    Else
        SafeCellText = Trim(CStr(cell.Value))
    End If

    If Len(SafeCellText) = 0 Then
        SafeCellText = Trim(CStr(cell.Text))
    End If

    Exit Function

ErrHandler:
    On Error Resume Next
    SafeCellText = Trim(CStr(cell.Text))
    On Error GoTo 0

End Function

Private Function ExtractFirstHttpUrl(ByVal text As String) As String

    Dim re As Object
    Dim matches As Object
    Dim s As String

    On Error GoTo ErrHandler

    If Len(text) = 0 Then
        ExtractFirstHttpUrl = ""
        Exit Function
    End If

    Set re = CreateObject("VBScript.RegExp")

    With re
        .Global = False
        .IgnoreCase = True
        .Pattern = "https?://[^\s""'<>）)】］、，,]+"
    End With

    Set matches = re.Execute(text)

    If matches.Count > 0 Then
        s = CStr(matches(0).Value)
        ExtractFirstHttpUrl = StripUrlTrailingMarks(s)
    Else
        ExtractFirstHttpUrl = ""
    End If

    Exit Function

ErrHandler:
    ExtractFirstHttpUrl = ""

End Function

Private Function NormalizeUrlCandidate(ByVal s As String) As String

    s = Trim(CStr(s))

    If Len(s) = 0 Then
        NormalizeUrlCandidate = ""
        Exit Function
    End If

    s = Replace(s, vbCr, "")
    s = Replace(s, vbLf, "")
    s = Replace(s, vbTab, "")
    s = Replace(s, """", "")

    s = Trim(s)

    If Left(s, 1) = "<" And Right(s, 1) = ">" Then
        s = Mid(s, 2, Len(s) - 2)
    End If

    s = StripUrlTrailingMarks(s)

    If IsHttpUrl(s) Then
        NormalizeUrlCandidate = s
    Else
        NormalizeUrlCandidate = ""
    End If

End Function

Private Function StripUrlTrailingMarks(ByVal s As String) As String

    Dim lastChar As String

    s = Trim(s)

    Do While Len(s) > 0

        lastChar = Right(s, 1)

        Select Case lastChar
            Case ".", ",", "、", "。", ")", "）", "]", "］", "】", ">", ";", "；"
                s = Left(s, Len(s) - 1)
            Case Else
                Exit Do
        End Select

    Loop

    StripUrlTrailingMarks = s

End Function

Private Function IsHttpUrl(ByVal s As String) As Boolean

    Dim u As String

    u = LCase(Trim(s))

    IsHttpUrl = _
        Left(u, 7) = "http://" Or _
        Left(u, 8) = "https://"

End Function

Private Sub StyleLinkInputRow(ByVal ws As Worksheet, ByVal rowNo As Long)

    On Error Resume Next

    With ws.Range("A" & rowNo & ":G" & rowNo)
        .VerticalAlignment = xlVAlignTop
        .WrapText = True
    End With

    ws.Cells(rowNo, "A").Interior.Color = RGB(255, 255, 204)
    ws.Cells(rowNo, "B").Interior.Color = RGB(235, 247, 255)
    ws.Range("C" & rowNo & ":G" & rowNo).Interior.Color = RGB(248, 248, 248)

    If Len(CStr(ws.Cells(rowNo, "B").Value)) > 0 Then

Private Function FindLatestRunFolder(ByVal workDir As String) As String

    Dim fso As Object
    Dim folder As Object
    Dim subFolder As Object

    Dim latestPath As String
    Dim latestDate As Date

    Set fso = CreateObject("Scripting.FileSystemObject")

    If Not fso.FolderExists(workDir) Then Exit Function

    Set folder = fso.GetFolder(workDir)

    For Each subFolder In folder.SubFolders

        If LCase(Left(CStr(subFolder.Name), 4)) = "run_" Then

            If Len(latestPath) = 0 Or subFolder.DateLastModified > latestDate Then
                latestPath = CStr(subFolder.Path)
                latestDate = subFolder.DateLastModified
            End If

        End If

    Next subFolder

    FindLatestRunFolder = latestPath

End Function

Private Function ImportResultsFromRunDir( _
        ByVal wsLinks As Worksheet, _
        ByVal runDir As String _
    ) As Long

    Dim fso As Object
    Dim folder As Object
    Dim file As Object

    Dim txt As String
    Dim lines() As String
    Dim parts() As String
    Dim line As String
    Dim i As Long
    Dim rowNo As Long

    Set fso = CreateObject("Scripting.FileSystemObject")

    If Not fso.FolderExists(runDir) Then Exit Function

    Set folder = fso.GetFolder(runDir)

    For Each file In folder.Files

        If LCase(Left(CStr(file.Name), 7)) = "result_" And LCase(Right(CStr(file.Name), 4)) = ".tsv" Then

            txt = ReadTextUtf8(CStr(file.Path))
            txt = Replace(txt, vbCrLf, vbLf)
            txt = Replace(txt, vbCr, vbLf)

            lines = Split(txt, vbLf)

            For i = LBound(lines) To UBound(lines)

                line = Trim(lines(i))

                If Len(line) > 0 Then

                    parts = Split(line, vbTab)

                    If UBound(parts) >= 6 Then

                        rowNo = CLng(Replace(parts(0), ChrW(&HFEFF), ""))

                        wsLinks.Cells(rowNo, "B").Value = parts(1)
                        wsLinks.Cells(rowNo, "C").Value = parts(2)
                        wsLinks.Cells(rowNo, "D").Value = parts(3)
                        wsLinks.Cells(rowNo, "E").Value = parts(4)
                        wsLinks.Cells(rowNo, "F").Value = parts(5)
                        wsLinks.Cells(rowNo, "G").Value = parts(6)

                        ImportResultsFromRunDir = ImportResultsFromRunDir + 1

                    End If

                End If

            Next i

        End If

    Next file

End Function

Public Sub BreakExternalUpdateLinksOnly()

    Dim links As Variant
    Dim i As Long
    Dim count As Long

    On Error GoTo ErrHandler

    links = ThisWorkbook.LinkSources(Type:=xlLinkTypeExcelLinks)

    If IsEmpty(links) Then
        MsgBox "外部更新リンクは見つかりませんでした。", vbInformation
        Exit Sub
    End If

    For i = LBound(links) To UBound(links)
        ThisWorkbook.BreakLink _
            Name:=CStr(links(i)), _
            Type:=xlLinkTypeExcelLinks
        count = count + 1
    Next i

    MsgBox "外部更新リンクを " & count & " 件解除しました。", vbInformation

    Exit Sub

ErrHandler:
    MsgBox "外部更新リンク解除中にエラーが発生しました: " & Err.Description, vbExclamation

End Sub

Private Function EnsureSheetByPosition( _
        ByVal indexNo As Long, _
        ByVal sheetName As String _
    ) As Worksheet

    Dim ws As Worksheet

    Do While ThisWorkbook.Worksheets.Count < indexNo
        ThisWorkbook.Worksheets.Add After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count)
    Loop

    Set ws = ThisWorkbook.Worksheets(indexNo)

    On Error Resume Next
    ws.Name = sheetName
    On Error GoTo 0

    Set EnsureSheetByPosition = ws

End Function

Private Function EnsureSheetByName(ByVal sheetName As String) As Worksheet

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(sheetName)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        ws.Name = sheetName
    End If

    Set EnsureSheetByName = ws

End Function

Private Function ResolveSupportFilePath( _
        ByVal rootDir As String, _
        ByVal fileName As String _
    ) As String

    Dim p As String
    Dim fso As Object

    Set fso = CreateObject("Scripting.FileSystemObject")

    If Len(ThisWorkbook.Path) > 0 Then
        p = CombinePath(ThisWorkbook.Path, fileName)
        If fso.FileExists(p) Then
            ResolveSupportFilePath = p
            Exit Function
        End If
    End If

    p = CombinePath(rootDir, fileName)
    If fso.FileExists(p) Then
        ResolveSupportFilePath = p
        Exit Function
    End If

    ResolveSupportFilePath = ""

End Function

Private Function NormalizeFolderPath(ByVal folderPath As String) As String

    folderPath = Trim(folderPath)
    folderPath = Replace(folderPath, """", "")

    Do While Len(folderPath) > 3 And _
        (Right(folderPath, 1) = "\" Or Right(folderPath, 1) = "/")
        folderPath = Left(folderPath, Len(folderPath) - 1)
    Loop

    NormalizeFolderPath = folderPath

End Function

Private Function CombinePath( _
        ByVal folderPath As String, _
        ByVal fileName As String _
    ) As String

    If Right(folderPath, 1) = "\" Or Right(folderPath, 1) = "/" Then
        CombinePath = folderPath & fileName
    Else
        CombinePath = folderPath & "\" & fileName
    End If

End Function

Private Sub EnsureFolderExists(ByVal folderPath As String)

    Dim fso As Object
    Dim parentPath As String

    Set fso = CreateObject("Scripting.FileSystemObject")

    folderPath = NormalizeFolderPath(folderPath)

    If Len(folderPath) = 0 Then Exit Sub
    If fso.FolderExists(folderPath) Then Exit Sub

    parentPath = fso.GetParentFolderName(folderPath)

    If Len(parentPath) > 0 Then
        If Not fso.FolderExists(parentPath) Then
            EnsureFolderExists parentPath
        End If
    End If

    If Not fso.FolderExists(folderPath) Then
        fso.CreateFolder folderPath
    End If

End Sub

Private Sub WriteTextUtf8( _
        ByVal filePath As String, _
        ByVal text As String _
    )

    Dim stream As Object

    Set stream = CreateObject("ADODB.Stream")

    With stream
        .Type = 2
        .Charset = "utf-8"
        .Open
        .WriteText text
        .SaveToFile filePath, 2
        .Close
    End With

End Sub

Private Function ReadTextUtf8(ByVal filePath As String) As String

    On Error GoTo ErrHandler

    Dim stream As Object

    Set stream = CreateObject("ADODB.Stream")

    With stream
        .Type = 2
        .Charset = "utf-8"
        .Open
        .LoadFromFile filePath
        ReadTextUtf8 = .ReadText
        .Close
    End With

    Exit Function

ErrHandler:
    ReadTextUtf8 = ""

End Function
