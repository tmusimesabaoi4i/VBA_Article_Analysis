Option Explicit

' ============================================================
' README・入力シート整形・リンククリーニング用モジュール
'
' 実行:
'   SetupDownloaderWorkbookReadmeAndLayout
'
' 補助実行:
'   CleanLinkSheetOnly
'   BreakExternalUpdateLinksOnly
'
' 前提:
'   1枚目: 設定シート
'     A1: プロキシアドレス
'     A2: 出力先ルートフォルダ
'     A3: 並列ワーカー数
'
'   2枚目: リンク一覧シート
'     A1: 見出し
'     A2以降: ハイパーリンク
'
' 注意:
'   READMEシートは3枚目以降に配置します。
'   既存の処理マクロが Worksheets(1), Worksheets(2) を参照していても壊れない構成です。
' ============================================================

Private Const SHEET_README_NAME As String = "README"

Private Const FIRST_LINK_ROW As Long = 2
Private Const DEFAULT_WORKER_COUNT As Long = 3
Private Const MAX_WORKER_COUNT As Long = 6

' True:
'   ブック内の外部更新リンクを解除します。
'   Sheet2の貼り付けで外部参照が混入する事故を避けたい場合は True 推奨。
Private Const BREAK_EXTERNAL_WORKBOOK_LINKS_ON_SETUP As Boolean = True

' 色
Private Const COLOR_HEADER As Long = &H404040
Private Const COLOR_INPUT As Long = &HE6FFFF
Private Const COLOR_NOTE As Long = &HF2F2F2
Private Const COLOR_WARN As Long = &HCCFFFF
Private Const COLOR_BLUE_TEXT As Long = &HC05000

' ============================================================
' メイン
' ============================================================

Public Sub SetupDownloaderWorkbookReadmeAndLayout()

    Dim wsCfg As Worksheet
    Dim wsList As Worksheet
    Dim wsReadme As Worksheet

    Dim proxyVal As String
    Dim rootVal As String
    Dim workerVal As String

    Dim linkTexts() As String
    Dim linkAddresses() As String
    Dim linkSubAddresses() As String
    Dim linkCount As Long

    Dim oldCalc As XlCalculation

    On Error GoTo ErrHandler

    oldCalc = Application.Calculation

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual
    Application.StatusBar = "README・入力形式を整備中..."

    ResolveMainSheets wsCfg, wsList

    proxyVal = CellValueAsText(wsCfg.Range("A1"))
    rootVal = CellValueAsText(wsCfg.Range("A2"))
    workerVal = CellValueAsText(wsCfg.Range("A3"))

    linkCount = CollectLinkRecords( _
                    wsList, _
                    linkTexts, _
                    linkAddresses, _
                    linkSubAddresses _
                )

    ResetWorksheetHard wsCfg
    ResetWorksheetHard wsList

    BuildConfigSheet wsCfg, proxyVal, rootVal, workerVal
    BuildLinkSheet wsList, linkTexts, linkAddresses, linkSubAddresses, linkCount

    Set wsReadme = EnsureReadmeSheet()
    ResetWorksheetHard wsReadme
    BuildReadmeSheet wsReadme

    EnsureReadmeAfterMainSheets wsReadme

    If BREAK_EXTERNAL_WORKBOOK_LINKS_ON_SETUP Then
        BreakExternalWorkbookLinks False
    End If

    wsCfg.Activate

    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox _
        "READMEと入力シートの整形が完了しました。" & vbCrLf & _
        "Sheet2 A列は、テキストとハイパーリンクのみ保持する形に整理済みです。", _
        vbInformation

    Exit Sub

ErrHandler:
    Application.StatusBar = False
    Application.Calculation = oldCalc
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "整形処理でエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' Sheet2 A列だけクリーニングしたい場合
' ============================================================

Public Sub CleanLinkSheetOnly()

    Dim wsCfg As Worksheet
    Dim wsList As Worksheet

    Dim linkTexts() As String
    Dim linkAddresses() As String
    Dim linkSubAddresses() As String
    Dim linkCount As Long

    On Error GoTo ErrHandler

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.DisplayAlerts = False

    ResolveMainSheets wsCfg, wsList

    linkCount = CollectLinkRecords( _
                    wsList, _
                    linkTexts, _
                    linkAddresses, _
                    linkSubAddresses _
                )

    ResetWorksheetHard wsList
    BuildLinkSheet wsList, linkTexts, linkAddresses, linkSubAddresses, linkCount

    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "Sheet2 A列を、テキストとハイパーリンクのみ保持する形に整理しました。", vbInformation

    Exit Sub

ErrHandler:
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "リンク整理でエラーが発生しました: " & Err.Description, vbExclamation

End Sub

' ============================================================
' 外部更新リンクだけ解除したい場合
' ============================================================

Public Sub BreakExternalUpdateLinksOnly()

    BreakExternalWorkbookLinks True

End Sub

' ============================================================
' メインシート解決
' ============================================================

Private Sub ResolveMainSheets( _
        ByRef wsCfg As Worksheet, _
        ByRef wsList As Worksheet _
    )

    Dim ws As Worksheet
    Dim foundCount As Long

    If ThisWorkbook.Worksheets.Count < 2 Then
        ThisWorkbook.Worksheets.Add After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count)
    End If

    foundCount = 0

    For Each ws In ThisWorkbook.Worksheets

        If ws.Name <> SHEET_README_NAME Then

            foundCount = foundCount + 1

            If foundCount = 1 Then
                Set wsCfg = ws
            ElseIf foundCount = 2 Then
                Set wsList = ws
                Exit For
            End If

        End If

    Next ws

    If wsCfg Is Nothing Then
        Set wsCfg = ThisWorkbook.Worksheets(1)
    End If

    If wsList Is Nothing Then
        Set wsList = ThisWorkbook.Worksheets(2)
    End If

    ' 既存の処理マクロが Worksheets(1), Worksheets(2) を使っていても壊れないようにする
    If wsCfg.Index <> 1 Then
        wsCfg.Move Before:=ThisWorkbook.Worksheets(1)
    End If

    If wsList.Index <> 2 Then
        wsList.Move After:=wsCfg
    End If

End Sub

Private Function EnsureReadmeSheet() As Worksheet

    Dim ws As Worksheet

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(SHEET_README_NAME)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        ws.Name = SHEET_README_NAME
    End If

    Set EnsureReadmeSheet = ws

End Function

Private Sub EnsureReadmeAfterMainSheets(ByVal wsReadme As Worksheet)

    If wsReadme.Index <= 2 Then
        wsReadme.Move After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count)
    End If

End Sub

' ============================================================
' リンク入力シート作成
' ============================================================

Private Sub BuildLinkSheet( _
        ByVal ws As Worksheet, _
        ByRef linkTexts() As String, _
        ByRef linkAddresses() As String, _
        ByRef linkSubAddresses() As String, _
        ByVal linkCount As Long _
    )

    Dim i As Long
    Dim r As Long
    Dim c As Range
    Dim displayText As String
    Dim addr As String
    Dim subAddr As String

    On Error Resume Next
    ws.Name = "02_LINKS"
    On Error GoTo 0

    ws.Range("A1").Value = "ダウンロード対象リンク"
    ws.Range("B1").Value = "URL"
    ws.Range("C1").Value = "ダウンロード先"
    ws.Range("D1").Value = "展開先"
    ws.Range("E1").Value = "PDF結果"
    ws.Range("F1").Value = "HTML結果"
    ws.Range("G1").Value = "処理結果"

    For i = 1 To linkCount

        r = FIRST_LINK_ROW + i - 1
        Set c = ws.Cells(r, "A")

        displayText = linkTexts(i)
        addr = linkAddresses(i)
        subAddr = linkSubAddresses(i)

        If Len(displayText) = 0 Then
            If Len(addr) > 0 Then
                displayText = addr
            ElseIf Len(subAddr) > 0 Then
                displayText = subAddr
            End If
        End If

        c.Value = displayText

        If Len(addr) > 0 Or Len(subAddr) > 0 Then
            ws.Hyperlinks.Add _
                Anchor:=c, _
                Address:=addr, _
                SubAddress:=subAddr, _
                TextToDisplay:=displayText
        End If

    Next i

    StyleHeaderRow ws.Range("A1:G1")

    ws.Columns("A").ColumnWidth = 60
    ws.Columns("B").ColumnWidth = 60
    ws.Columns("C").ColumnWidth = 46
    ws.Columns("D").ColumnWidth = 46
    ws.Columns("E").ColumnWidth = 34
    ws.Columns("F").ColumnWidth = 34
    ws.Columns("G").ColumnWidth = 34

    ws.Columns("A:G").WrapText = True
    ws.Rows(1).RowHeight = 28

    ws.Range("A:A").Interior.Color = RGB(255, 255, 204)
    ws.Range("B:G").Interior.Color = RGB(248, 248, 248)

    ws.Range("A1:G1").AutoFilter

    If linkCount > 0 Then
        ws.Range("A2:A" & FIRST_LINK_ROW + linkCount - 1).Font.Color = RGB(5, 99, 193)
        ws.Range("A2:A" & FIRST_LINK_ROW + linkCount - 1).Font.Underline = xlUnderlineStyleSingle
        ws.Range("A2:G" & FIRST_LINK_ROW + linkCount - 1).Borders.LineStyle = xlContinuous
        ws.Range("A2:G" & FIRST_LINK_ROW + linkCount - 1).VerticalAlignment = xlVAlignTop
    End If

    ws.Range("I1").Value = "入力ルール"
    ws.Range("I2").Value = "A2以降に、ダウンロード対象のハイパーリンクを貼り付けてください。"
    ws.Range("I3").Value = "貼り付け後、CleanLinkSheetOnly を実行すると、書式・数式・外部更新リンクを除去し、テキストとハイパーリンクのみ保持します。"
    ws.Range("I4").Value = "B〜G列は処理結果の出力欄です。手入力不要です。"
    ws.Range("I1:I4").WrapText = True
    ws.Range("I1").Font.Bold = True
    ws.Range("I1:I4").Interior.Color = RGB(242, 242, 242)
    ws.Columns("I").ColumnWidth = 60

    AddCellNote ws.Range("A1"), "A2以降にリンクを貼り付けます。貼り付け後に CleanLinkSheetOnly を実行すると、不要な書式や更新リンクが除去されます。"

    FreezeTopRow ws

End Sub

' ============================================================
' リンク情報の保持・抽出
' ============================================================

Private Function CollectLinkRecords( _
        ByVal ws As Worksheet, _
        ByRef texts() As String, _
        ByRef addresses() As String, _
        ByRef subAddresses() As String _
    ) As Long

    Dim lastRow As Long
    Dim r As Long
    Dim n As Long

    Dim displayText As String
    Dim addr As String
    Dim subAddr As String

    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    If lastRow < FIRST_LINK_ROW Then
        CollectLinkRecords = 0
        Exit Function
    End If

    For r = FIRST_LINK_ROW To lastRow

        displayText = CellValueAsText(ws.Cells(r, "A"))
        addr = ""
        subAddr = ""

        ExtractHyperlinkInfo ws.Cells(r, "A"), displayText, addr, subAddr

        If Len(addr) = 0 And Len(subAddr) = 0 Then
            If IsHttpUrl(displayText) Then
                addr = displayText
            End If
        End If

        If Len(displayText) > 0 Or Len(addr) > 0 Or Len(subAddr) > 0 Then

            n = n + 1

            ReDim Preserve texts(1 To n)
            ReDim Preserve addresses(1 To n)
            ReDim Preserve subAddresses(1 To n)

            texts(n) = displayText
            addresses(n) = addr
            subAddresses(n) = subAddr

        End If

    Next r

    CollectLinkRecords = n

End Function

Private Sub ExtractHyperlinkInfo( _
        ByVal cell As Range, _
        ByRef displayText As String, _
        ByRef address As String, _
        ByRef subAddress As String _
    )

    Dim tmpAddress As String
    Dim tmpDisplay As String

    On Error Resume Next

    If cell.Hyperlinks.Count > 0 Then

        address = cell.Hyperlinks(1).Address
        subAddress = cell.Hyperlinks(1).SubAddress

        If Len(displayText) = 0 Then
            displayText = cell.Hyperlinks(1).TextToDisplay
        End If

    End If

    On Error GoTo 0

    If Len(address) = 0 Then

        If cell.HasFormula Then

            tmpAddress = ""
            tmpDisplay = ""

            If TryParseHyperlinkFormula(CStr(cell.Formula), tmpAddress, tmpDisplay) Then

                address = tmpAddress

                If Len(displayText) = 0 And Len(tmpDisplay) > 0 Then
                    displayText = tmpDisplay
                End If

            End If

        End If

    End If

    If Len(address) = 0 And IsHttpUrl(displayText) Then
        address = displayText
    End If

End Sub

Private Function TryParseHyperlinkFormula( _
        ByVal formulaText As String, _
        ByRef address As String, _
        ByRef textToDisplay As String _
    ) As Boolean

    On Error GoTo ErrHandler

    Dim re As Object
    Dim matches As Object

    If InStr(1, formulaText, "HYPERLINK", vbTextCompare) = 0 Then
        TryParseHyperlinkFormula = False
        Exit Function
    End If

    Set re = CreateObject("VBScript.RegExp")

    With re
        .Global = True
        .IgnoreCase = True
        .Pattern = """([^""]*)"""
    End With

    Set matches = re.Execute(formulaText)

    If matches.Count >= 1 Then
        address = CStr(matches(0).SubMatches(0))

        If matches.Count >= 2 Then
            textToDisplay = CStr(matches(1).SubMatches(0))
        Else
            textToDisplay = address
        End If

        TryParseHyperlinkFormula = True
    Else
        TryParseHyperlinkFormula = False
    End If

    Exit Function

ErrHandler:
    TryParseHyperlinkFormula = False

End Function

' ============================================================

Private Sub BreakExternalWorkbookLinks(ByVal showMessage As Boolean)

    On Error GoTo ErrHandler

    Dim links As Variant
    Dim i As Long
    Dim count As Long

    links = ThisWorkbook.LinkSources(Type:=xlLinkTypeExcelLinks)

    If IsEmpty(links) Then
        If showMessage Then
            MsgBox "外部更新リンクは見つかりませんでした。", vbInformation
        End If
        Exit Sub
    End If

    For i = LBound(links) To UBound(links)
        ThisWorkbook.BreakLink _
            Name:=CStr(links(i)), _
            Type:=xlLinkTypeExcelLinks
        count = count + 1
    Next i

    If showMessage Then
        MsgBox "外部更新リンクを " & count & " 件解除しました。", vbInformation
    End If

    Exit Sub

ErrHandler:
    If showMessage Then
        MsgBox "外部更新リンクの解除中にエラーが発生しました: " & Err.Description, vbExclamation
    End If

End Sub

Private Function CellValueAsText(ByVal cell As Range) As String

    On Error GoTo ErrHandler

    If IsError(cell.Value) Then
        CellValueAsText = ""
    Else
        CellValueAsText = Trim(CStr(cell.Value))
    End If

    Exit Function

ErrHandler:
    CellValueAsText = Trim(CStr(cell.Text))

End Function

Private Function IsHttpUrl(ByVal s As String) As Boolean

    Dim u As String

    u = LCase(Trim(s))

    IsHttpUrl = _
        Left(u, 7) = "http://" Or _
        Left(u, 8) = "https://"

End Function
